cmake_minimum_required(VERSION 3.20)
project(ValtCommandEmbeddingExamples LANGUAGES CXX)
include(CTest)
find_package(ValtCommand CONFIG REQUIRED)

foreach(example running results native debugger known_limitations reference)
  add_executable(${example} ${example}.cpp)
  target_compile_features(${example} PRIVATE cxx_std_20)
  target_link_libraries(${example} PRIVATE ValtCommand::valtcommand)
  add_test(NAME ${example} COMMAND ${example})
endforeach()
add_executable(host_file host_file.cpp)
target_compile_features(host_file PRIVATE cxx_std_20)
target_link_libraries(host_file PRIVATE ValtCommand::valtcommand)
add_test(NAME host_file COMMAND host_file ${CMAKE_CURRENT_SOURCE_DIR}/sample.vlt)
